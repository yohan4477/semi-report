## Global Constraints

- 대상 파일: `insights/clusters/*.md` 9개. frontmatter(`sources`·`source_hashes`·`source_dates`·`as_of`·`covered_through`)는 **수정 금지**.
- 블록 제목·순서 고정: `## 통합 논지` → `## 공통 진단` → `## 연결·인과` → `## 상충·이견` → `## 아직 모르는 것` → `## 함의·다음 확인 포인트`.
- 앵커 형식: `(문서약칭 YY-MM-DD)`. 약칭은 문서 제목에서 `[YYMMDD]`를 뗀 뒤 12자 이내.
- 인정 수치: 단위·연도가 붙은 것만. `3사`·`2가지`처럼 단위 없는 수는 R1을 통과시키지 않는다.
- 금지 표현(R5): `가속`, `확대`, `강화`, `진화하고 있다`, `흐름은 일관`, `확신도 중간`, `중요하다`, `주목된다`, `전망이다` — 같은 불릿에 인정 수치가 없을 때만 실패.
- 목표 지표: 숫자 포함 ≥80%, 앵커 100%, 클러스터당 `아직 모르는 것` 항목 ≥1.
- 문체(R7, 자동 검사 불가): **현상을 먼저 쓰고 용어는 괄호로 뒤에** 붙인다(`물을 분당 4리터 넘게 흘려도 더 안 식는다 — 냉각판 방식(DLC)의 한계`, `DLC는 4LPM에서 포화한다`가 아니라). 회사는 정체를 한 마디로. 각 불릿은 수치 + 함의로 끝냄. 초보자용 비유·서사 금지.
- 원문에서 확인하지 않은 수치는 쓰지 않는다. 원문에 없는 계산·환산을 새로 만들지 않는다.
- 인코딩: 모든 파일 IO는 `io.open(..., encoding='utf-8')`. Windows 콘솔 출력 시 `PYTHONIOENCODING=utf-8`.

# 인사이트 문장 구체성 규칙 — 블록 파싱 + R1~R6 판정
import re

BLOCKS = ['통합 논지', '공통 진단', '연결·인과', '상충·이견', '아직 모르는 것', '함의·다음 확인 포인트']


def parse(body):
    """## 블록 → [(줄번호, 텍스트)]. 불릿은 '- ' 제거, 산문은 줄 그대로."""
    out, cur = {}, None
    for i, line in enumerate(body.splitlines(), 1):
        h = re.match(r'^##\s+(.+?)\s*$', line)
        if h:
            cur = h.group(1)
            out[cur] = []
            continue
        if cur is None:
            continue
        s = line.strip()
        if not s or s.startswith('#') or s.startswith('---'):
            continue
        m = re.match(r'^-\s+(.*)$', s)
        out[cur].append((i, m.group(1).strip() if m else s))
    return out
```

BANNED = ['가속', '확대', '강화', '진화하고 있다', '흐름은 일관', '확신도 중간',
          '중요하다', '주목된다', '전망이다']

NEED_NUM = ('공통 진단', '상충·이견', '아직 모르는 것')
NEED_ANCHOR = ('공통 진단', '연결·인과', '상충·이견', '아직 모르는 것', '함의·다음 확인 포인트')
UNKNOWN = '아직 모르는 것'
THESIS = '통합 논지'


def check_body(body):
    blocks = parse(body)
    out = []

    def add(line, rule, text):
        out.append({'line': line, 'rule': rule, 'level': 'FAIL', 'text': text[:70]})

    for name, units in blocks.items():
        for line, text in units:
            bare = strip_anchors(text)
            if name in NEED_NUM and not has_number(bare):
                add(line, 'R1', text)
            if name in NEED_ANCHOR and not has_anchor(text):
                add(line, 'R2', text)
            if name in NEED_ANCHOR and not has_number(bare):
                for w in BANNED:
                    if w in bare:
                        add(line, 'R5', text)
                        break

    if not blocks.get(UNKNOWN):
        add(0, 'R3', '아직 모르는 것 블록 없음 또는 항목 0개')

    thesis = blocks.get(THESIS) or []
    joined = ' '.join(t for _, t in thesis)
    bare = strip_anchors(joined)
    sentences = [s for s in re.split(r'(?<=[.!?])\s+', bare.strip()) if s]
    nums = len(NUM.findall(bare))
    if not thesis:
        add(0, 'R4', '통합 논지 없음')
    elif len(sentences) > 2 or nums < 2:
        add(thesis[0][0], 'R4', '문장 %d개·수치 %d개 (2문장 이하·수치 2개 이상 필요)' % (len(sentences), nums))

def _norm(s):
    return s.replace(',', '').replace(' ', '')


def numbers(text):
    return [_norm(m.group(0)) for m in NUM.finditer(text)]


def check_r6(body, source_texts):
    hay = _norm(' '.join(source_texts))
    out = []
    for name, units in parse(body).items():
        for line, text in units:
            for tok in numbers(strip_anchors(text)):
                if re.fullmatch(r'(19|20)\d\d', tok):
                    continue  # 연도는 서술 변형이 잦아 검사 제외
                if tok not in hay:
                    out.append({'line': line, 'rule': 'R6', 'level': 'WARN',
                                'text': '원문에 없음: %s — %s' % (tok, text[:50])})
    return out

import coverage as cov

def validate(manifest, clusters):
    ids = {s['id'] for s in manifest['sources']}
    errs = []
    for c in clusters:
        if not c['cluster_id']:
            errs.append('cluster_id 없음(frontmatter 깨짐)')
        for sid in c['sources']:
            if sid not in ids:
                errs.append('%s: 존재하지 않는 소스 %s' % (c['cluster_id'] or '?', sid))
    return errs


def _body(text):
    m = re.match(r'^---\n.*?\n---\n(.*)$', text, re.DOTALL)
    return m.group(1) if m else text


def _sources(cluster_text, man):
    c = cov.parse_cluster(cluster_text)
    out = []
    for sid in c['sources']:
        s = man.get(sid)
        if not s:
            continue
        p = os.path.join(ROOT, s['path'])
        if os.path.exists(p):
            out.append(io.open(p, encoding='utf-8').read())
    return out


def main():
    manifest = json.load(io.open(MAN_PATH, encoding='utf-8'))
    man = {s['id']: s for s in manifest['sources']}
    errs = validate(manifest, cov.load_clusters())
    for e in errs:
        print('ERROR %s' % e)

    fails = warns = units = with_num = with_anchor = 0
    for path in sorted(glob.glob(os.path.join(ROOT, 'insights', 'clusters', '*.md'))):
        text = io.open(path, encoding='utf-8').read()
        body = _body(text)
        rel = os.path.relpath(path, ROOT)
        findings = sp.check_body(body) + sp.check_r6(body, _sources(text, man))
        for f in findings:
            print('%s %s:%d [%s] %s' % (f['level'], rel, f['line'], f['rule'], f['text']))
            if f['level'] == 'FAIL':
                fails += 1
            else:
                warns += 1
        for name, us in sp.parse(body).items():
            for _, t in us:
                units += 1
                if sp.has_number(sp.strip_anchors(t)):
                    with_num += 1
                if sp.has_anchor(t):
                    with_anchor += 1

    pct = lambda a, b: (100 * a // b) if b else 0
    print('요약: 문장 %d / FAIL %d / WARN %d / 숫자 포함 %d%% / 앵커 %d%%'
          % (units, fails, warns, pct(with_num, units), pct(with_anchor, units)))
    sys.exit(1 if (errs or fails) else 0)


if __name__ == '__main__':
    main()
```
